Скелет класса

Вот полная структура global_planner_node.cpp — всё, что мы написали в шагах 2-5, собрано воедино:

#include <rclcpp/rclcpp.hpp>
#include <sensor_msgs/msg/laser_scan.hpp>
#include <nav_msgs/msg/odometry.hpp>
#include <nav_msgs/msg/occupancy_grid.hpp>
#include <nav_msgs/msg/path.hpp>
#include <geometry_msgs/msg/twist.hpp>
#include <geometry_msgs/msg/pose_stamped.hpp>
#include <tf2/utils.h>
#include <cmath>
#include <queue>
#include <vector>
#include <algorithm>

using namespace std::chrono_literals;

class GlobalPlanner : public rclcpp::Node {
public:
    GlobalPlanner() : Node("global_planner_node") {
        // Subscribers
        scan_sub_ = create_subscription<sensor_msgs::msg::LaserScan>(
            "/scan", 10, [this](auto msg){ scanCallback(msg); });
        odom_sub_ = create_subscription<nav_msgs::msg::Odometry>(
            "/odom", 10, [this](auto msg){ odomCallback(msg); });
        goal_sub_ = create_subscription<geometry_msgs::msg::PoseStamped>(
            "/goal_pose", 10, [this](auto msg){ goalCallback(msg); });

        // Publishers
        cmd_pub_  = create_publisher<geometry_msgs::msg::Twist>("/cmd_vel", 10);
        map_pub_  = create_publisher<nav_msgs::msg::OccupancyGrid>("/map", 10);
        path_pub_ = create_publisher<nav_msgs::msg::Path>("/global_plan", 10);

        // Таймер управления (10 Hz)
        timer_ = create_wall_timer(100ms, [this](){ controlLoop(); });

        RCLCPP_INFO(get_logger(), "Global Planner запущен!");
    }

private:
    // === Параметры карты ===
    static const int MAP_SIZE = 200;
    static constexpr double RESOLUTION = 0.05;
    static constexpr double ORIGIN_X = -5.0;
    static constexpr double ORIGIN_Y = -5.0;

    // === Состояние ===
    std::vector<int8_t> map_data_{MAP_SIZE * MAP_SIZE, -1};
    double robot_x_ = 0, robot_y_ = 0, robot_yaw_ = 0;
    double goal_x_ = 0, goal_y_ = 0;
    bool has_goal_ = false;
    std::vector<std::pair<int,int>> current_path_;

    // === ROS2 объекты ===
    rclcpp::Subscription<sensor_msgs::msg::LaserScan>::SharedPtr scan_sub_;
    rclcpp::Subscription<nav_msgs::msg::Odometry>::SharedPtr odom_sub_;
    rclcpp::Subscription<geometry_msgs::msg::PoseStamped>::SharedPtr goal_sub_;
    rclcpp::Publisher<geometry_msgs::msg::Twist>::SharedPtr cmd_pub_;
    rclcpp::Publisher<nav_msgs::msg::OccupancyGrid>::SharedPtr map_pub_;
    rclcpp::Publisher<nav_msgs::msg::Path>::SharedPtr path_pub_;
    rclcpp::TimerBase::SharedPtr timer_;

    // === Callbacks (шаг 2) ===
    void odomCallback(/*...*/);
    void scanCallback(/*...*/);
    void goalCallback(/*...*/);

    // === Карта (шаг 3) ===
    int worldToGrid(double coord, double origin);
    void updateMapWithLaser(double rx, double ry, double hx, double hy);
    void publishMap();
    bool isFree(int x, int y);

    // === A* (шаг 4) ===
    double heuristic(int x1, int y1, int x2, int y2);
    std::vector<std::pair<int,int>> planAStar(int sx, int sy, int gx, int gy);
    void replan();
    void publishPath();

    // === Pure Pursuit (шаг 5) ===
    double normalizeAngle(double a);
    void purePursuitControl();

    // === Главный цикл ===
    void controlLoop() {
        if (has_goal_) {
            replan();
            purePursuitControl();
        }
    }
};

int main(int argc, char** argv) {
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<GlobalPlanner>());
    rclcpp::shutdown();
    return 0;
}

Launch-файл

Создай launch/planner.launch.py — он запускает Gazebo, RViz и нашу ноду одной командой:

# launch/planner.launch.py
from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch_ros.actions import Node
from ament_index_python.packages import get_package_share_directory
import os

def generate_launch_description():
    # Запускаем Gazebo с TurtleBot3
    gazebo = IncludeLaunchDescription(
        PythonLaunchDescriptionSource(
            os.path.join(
                get_package_share_directory('turtlebot3_gazebo'),
                'launch', 'turtlebot3_world.launch.py'
            )
        )
    )

    # Наша нода
    planner = Node(
        package='global_planner',
        executable='global_planner_node',
        name='global_planner_node',
        output='screen'
    )

    # RViz
    rviz = Node(
        package='rviz2',
        executable='rviz2',
        name='rviz2',
        output='screen'
    )

    return LaunchDescription([gazebo, planner, rviz])

Добавь следующие отображения:

Отладка: если что-то не работает

Робот не двигается

  • Проверь: ros2 topic echo /cmd_vel --once — публикуется ли что-то?
  • Если нули — нода не отправляет команды. Проверь: есть ли has_goal_ = true?
  • Если числа ненулевые — проблема в Gazebo. Перезапусти.

Карта не появляется в RViz

  • ros2 topic echo /map --once — публикуется ли карта?
  • Проверь frame_id — должен быть "odom".
  • В RViz: Fixed Frame тоже должен быть odom.

A* не находит путь

  • Цель может быть за пределами карты → проверь ORIGIN_X/Y и MAP_SIZE.
  • Цель может стоять на стене → попробуй другую точку.
  • Добавь лог: RCLCPP_INFO(get_logger(), "A*: start(%d,%d) goal(%d,%d)", sx,sy,gx,gy);

Робот осциллирует / кружит

  • Увеличь LOOKAHEAD_DIST (0.3 → 0.5)
  • Уменьши коэффициент поворота (2.0 → 1.0)
  • Увеличь GOAL_TOLERANCE (0.15 → 0.25)
